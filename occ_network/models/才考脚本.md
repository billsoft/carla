"""
OccNetV3 TBPTT 训练脚本

TBPTT (Truncated Backpropagation Through Time) 训练模式：
- 累积多个时间步的 loss 后再 backward
- 允许梯度回传到历史帧的特征提取器
- 提升动态物体检测和时序一致性

与 train.py 的区别：
- train.py: 每个 batch 独立，历史帧全部 detach
- train_tbptt.py: 累积 tbptt_window 个 batch 后才 backward

使用方法：
    python train_tbptt.py --dataset /path/to/data --tbptt-window 3 --amp

注意事项：
1. 显存占用约为普通训练的 1.5~2 倍
2. 建议先用 train.py 预训练，再用 train_tbptt.py 微调
3. 如果 OOM，减小 --tbptt-window 或关闭 --amp
"""

import torch
import torch.nn as nn
import os
import time
import argparse
from torch.cuda.amp import GradScaler, autocast
from configs.default import config
from models import build_model
from losses.losses import OccLoss
from data.dataset import build_dataloader, FP16DataPrefetcher


def detach_temporal_history(model):
    """
    分离时序模块的所有历史帧，切断与旧计算图的连接
    在每个 TBPTT 窗口结束后调用
    """
    temporal = model.temporal
    if hasattr(temporal, 'fusion'):
        temporal = temporal.fusion
    
    # 分离所有历史 BEV
    for i in range(len(temporal.history_bevs)):
        if temporal.history_bevs[i] is not None:
            if hasattr(temporal.history_bevs[i], 'requires_grad') and temporal.history_bevs[i].requires_grad:
                temporal.history_bevs[i] = temporal.history_bevs[i].detach()
    
    # 分离所有历史 pose
    for i in range(len(temporal.history_poses)):
        if temporal.history_poses[i] is not None:
            if hasattr(temporal.history_poses[i], 'requires_grad') and temporal.history_poses[i].requires_grad:
                temporal.history_poses[i] = temporal.history_poses[i].detach()


def train_one_epoch_tbptt(model, loader, optimizer, scaler, loss_fn, epoch, config, tbptt_window=3):
    """
    TBPTT 训练一个 epoch
    
    核心逻辑：
    1. 累积 tbptt_window 个 batch 的 loss
    2. 一次性 backward（梯度可以回传到历史帧）
    3. optimizer.step()
    4. detach 所有历史，开始新窗口
    """
    model.train()
    total_loss = 0
    device = next(model.parameters()).device
    
    print(f"  [TBPTT] Window size: {tbptt_window}")
    print(f"  Creating prefetcher...", flush=True)
    prefetcher = FP16DataPrefetcher(loader, device)
    batch = prefetcher.next()
    
    step = 0
    window_step = 0  # 当前窗口内的 step
    accumulated_loss = torch.tensor(0.0, device=device)
    window_losses = []  # 记录窗口内每个 batch 的 loss
    
    start_time = time.time()
    last_scene_id = None
    
    while batch is not None:
        data_time = time.time() - start_time
        
        # 场景切换检测
        scene_id = batch.get('scene_id', None)
        timestamp = batch.get('timestamp', None)
        
        # 检测场景切换 - 如果切换了，结束当前窗口并 reset
        scene_changed = False
        if scene_id is not None and last_scene_id is not None:
            current_scene = scene_id[0] if isinstance(scene_id, (list, tuple)) else scene_id
            if current_scene != last_scene_id:
                scene_changed = True
                print(f"  [Scene Switch] {last_scene_id} -> {current_scene}")
        
        if scene_id is not None:
            last_scene_id = scene_id[0] if isinstance(scene_id, (list, tuple)) else scene_id
        
        # 如果场景切换，先处理累积的 loss，然后 reset
        if scene_changed and window_step > 0:
            # 处理当前窗口
            if accumulated_loss.requires_grad:
                if config.use_amp:
                    scaler.scale(accumulated_loss).backward()
                    if config.grad_clip > 0:
                        scaler.unscale_(optimizer)
                        nn.utils.clip_grad_norm_(model.parameters(), config.grad_clip)
                    scaler.step(optimizer)
                    scaler.update()
                else:
                    accumulated_loss.backward()
                    if config.grad_clip > 0:
                        nn.utils.clip_grad_norm_(model.parameters(), config.grad_clip)
                    optimizer.step()
            
            optimizer.zero_grad()
            detach_temporal_history(model)
            model.reset_temporal()
            
            accumulated_loss = torch.tensor(0.0, device=device)
            window_losses = []
            window_step = 0
        
        # Forward pass
        with autocast(enabled=config.use_amp):
            outputs = model(
                batch['images'],
                batch.get('ego_motion'),
                batch.get('ego_pose'),
                timestamp=timestamp,
                scene_id=scene_id,
                intrinsics=batch.get('intrinsics'),
                extrinsics=batch.get('extrinsics')
            )
            losses = loss_fn(outputs, batch)
            loss = losses['total']
        
        # NaN 检查
        if torch.isnan(loss) or torch.isinf(loss):
            print(f"Warning: Loss is {loss.item()} at Epoch {epoch} Step {step}, skipping batch.")
            batch = prefetcher.next()
            step += 1
            start_time = time.time()
            continue
        
        # 累积 loss
        accumulated_loss = accumulated_loss + loss
        window_losses.append(loss.item())
        window_step += 1
        total_loss += loss.item()
        
        # 检查是否到达窗口边界
        window_complete = (window_step >= tbptt_window)
        
        if window_complete:
            # 🔑 TBPTT 核心：一次性 backward 累积的 loss
            if config.use_amp:
                scaler.scale(accumulated_loss).backward()
                if config.grad_clip > 0:
                    scaler.unscale_(optimizer)
                    nn.utils.clip_grad_norm_(model.parameters(), config.grad_clip)
                scaler.step(optimizer)
                scaler.update()
            else:
                accumulated_loss.backward()
                if config.grad_clip > 0:
                    nn.utils.clip_grad_norm_(model.parameters(), config.grad_clip)
                optimizer.step()
            
            optimizer.zero_grad()
            
            # 🔑 窗口结束后，detach 所有历史
            detach_temporal_history(model)
            
            # 重置累积
            accumulated_loss = torch.tensor(0.0, device=device)
            window_losses = []
            window_step = 0
        
        batch_time = time.time() - start_time
        
        # 日志
        if step % config.log_interval == 0:
            mem = torch.cuda.max_memory_allocated() / 1024**3
            dist_loss = losses.get('distance', 0)
            if isinstance(dist_loss, torch.Tensor):
                dist_loss = dist_loss.item()
            depth_loss = losses.get('depth', 0)
            if isinstance(depth_loss, torch.Tensor):
                depth_loss = depth_loss.item()
            
            window_info = f"W:{window_step}/{tbptt_window}"
            print(f"Epoch {epoch} Step {step} {window_info} Loss: {loss.item():.4f} "
                  f"Focal: {losses.get('focal', 0):.4f} Dice: {losses.get('dice', 0):.4f} "
                  f"Dist: {dist_loss:.4f} Depth: {depth_loss:.4f} "
                  f"GPU: {mem:.2f}GB Data: {data_time:.3f}s Batch: {batch_time:.3f}s", flush=True)
        
        batch = prefetcher.next()
        step += 1
        start_time = time.time()
    
    # 处理最后一个不完整的窗口
    if window_step > 0 and accumulated_loss.requires_grad:
        if config.use_amp:
            scaler.scale(accumulated_loss).backward()
            if config.grad_clip > 0:
                scaler.unscale_(optimizer)
                nn.utils.clip_grad_norm_(model.parameters(), config.grad_clip)
            scaler.step(optimizer)
            scaler.update()
        else:
            accumulated_loss.backward()
            if config.grad_clip > 0:
                nn.utils.clip_grad_norm_(model.parameters(), config.grad_clip)
            optimizer.step()
        
        optimizer.zero_grad()
        detach_temporal_history(model)
    
    return total_loss / max(step, 1)


@torch.no_grad()
def validate(model, loader, loss_fn, config):
    """验证（与 train.py 相同）"""
    model.eval()
    total_loss = 0
    device = next(model.parameters()).device
    
    for batch in loader:
        for key in batch:
            if isinstance(batch[key], torch.Tensor):
                batch[key] = batch[key].to(device)
        
        with autocast(enabled=config.use_amp):
            outputs = model(batch['images'])
            losses = loss_fn(outputs, batch)
        
        total_loss += losses['total'].item()
    
    return total_loss / len(loader)


def main():
    parser = argparse.ArgumentParser(description='Train OccNetV3 with TBPTT')
    
    # 数据
    parser.add_argument('--dataset', type=str, default='./data', help='数据集根目录')
    
    # TBPTT 特有参数
    parser.add_argument('--tbptt-window', type=int, default=3, 
                        help='TBPTT 窗口大小（累积多少个 batch 后 backward）')
    
    # 训练参数
    parser.add_argument('--epochs', type=int, default=None, help='训练轮数')
    parser.add_argument('--batch-size', type=int, default=None, help='批量大小')
    parser.add_argument('--lr', type=float, default=None, help='学习率')
    parser.add_argument('--amp', action='store_true', default=None, help='启用混合精度训练')
    parser.add_argument('--no-amp', dest='amp', action='store_false', help='禁用混合精度训练')
    parser.add_argument('--grad-clip', type=float, default=None, help='梯度裁剪阈值')
    parser.add_argument('--num-workers', type=int, default=None)
    
    # 其他参数
    parser.add_argument('--save-dir', type=str, default=None, help='保存目录')
    parser.add_argument('--resume', type=str, default=None)
    parser.add_argument('--log-interval', type=int, default=None)
    
    args = parser.parse_args()
    
    # 更新 config
    config.data_dir = args.dataset
    
    # 🔑 强制启用 TBPTT 相关配置
    config.tbptt_steps = args.tbptt_window  # 传递给模型
    
    if args.batch_size:
        config.batch_size = args.batch_size
    if args.epochs:
        config.max_epochs = args.epochs
    if args.amp is not None:
        config.use_amp = args.amp
    if args.lr:
        config.lr = args.lr
    if args.grad_clip:
        config.grad_clip = args.grad_clip
    if args.num_workers:
        config.num_workers = args.num_workers
    if args.save_dir:
        config.save_dir = args.save_dir
    else:
        config.save_dir = './checkpoints_tbptt'  # 默认不同的目录
    if args.log_interval:
        config.log_interval = args.log_interval
    
    os.makedirs(config.save_dir, exist_ok=True)
    
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    
    # 构建模型
    model = build_model(config).to(device)
    loss_fn = OccLoss(config)
    
    # 优化器 - TBPTT 建议使用较小的学习率
    effective_lr = config.lr * 0.5  # TBPTT 学习率减半
    optimizer = torch.optim.AdamW(model.parameters(), lr=effective_lr, weight_decay=config.weight_decay)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=config.max_epochs)
    scaler = GradScaler(enabled=config.use_amp)
    
    start_epoch = 0
    if args.resume:
        ckpt = torch.load(args.resume, map_location=device)
        model.load_state_dict(ckpt['model'])
        optimizer.load_state_dict(ckpt['optimizer'])
        start_epoch = ckpt['epoch'] + 1
        print(f"Resumed from epoch {start_epoch}")
    
    train_loader = build_dataloader(config, 'train')
    val_loader = build_dataloader(config, 'val')
    
    print("=" * 60)
    print("OccNetV3 TBPTT Training")
    print("=" * 60)
    print(f"Device: {device}")
    print(f"AMP: {config.use_amp}")
    print(f"TBPTT Window: {args.tbptt_window}")
    print(f"Effective LR: {effective_lr} (base: {config.lr})")
    print(f"Train loader size: {len(train_loader)}")
    print(f"Val loader size: {len(val_loader)}")
    print("=" * 60)
    
    print("\n⚠️  TBPTT 训练注意事项:")
    print("   1. 显存占用约为普通训练的 1.5~2 倍")
    print("   2. 如果 OOM，减小 --tbptt-window")
    print("   3. 建议先用 train.py 预训练后再微调")
    print()
    
    best_loss = float('inf')
    
    for epoch in range(start_epoch, config.max_epochs):
        print(f"\nStarting epoch {epoch}...", flush=True)
        
        # 学习率预热
        if epoch < config.warmup_epochs:
            lr_scale = (epoch + 1) / config.warmup_epochs
            for pg in optimizer.param_groups:
                pg['lr'] = effective_lr * lr_scale
        
        # 重置时序状态
        model.reset_temporal()
        
        # TBPTT 训练
        train_loss = train_one_epoch_tbptt(
            model, train_loader, optimizer, scaler, loss_fn, 
            epoch, config, tbptt_window=args.tbptt_window
        )
        
        if epoch >= config.warmup_epochs:
            scheduler.step()
        
        print(f"Epoch {epoch} Train Loss: {train_loss:.4f}", flush=True)
        
        # 验证
        if (epoch + 1) % config.eval_interval == 0:
            model.reset_temporal()
            val_loss = validate(model, val_loader, loss_fn, config)
            print(f"Epoch {epoch} Val Loss: {val_loss:.4f}")
            
            if val_loss < best_loss:
                best_loss = val_loss
                torch.save({
                    'model': model.state_dict(),
                    'optimizer': optimizer.state_dict(),
                    'epoch': epoch,
                    'loss': val_loss,
                    'tbptt_window': args.tbptt_window,
                }, os.path.join(config.save_dir, 'best_tbptt.pth'))
                print(f"  ✅ Best model saved!")
        
        # 定期保存
        if (epoch + 1) % config.save_interval == 0:
            torch.save({
                'model': model.state_dict(),
                'optimizer': optimizer.state_dict(),
                'epoch': epoch,
                'tbptt_window': args.tbptt_window,
            }, os.path.join(config.save_dir, f'epoch_{epoch}_tbptt.pth'))
    
    print("\nTBPTT Training complete!")


if __name__ == '__main__':
    main()
