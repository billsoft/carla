  @torch.no_grad()
  def inference_with_uncertainty(model, config, device, images=None, num_samples=None, temperature=None):
      if images is None:
          images = torch.randn(1, config.num_cameras, config.in_channels, *config.image_size, dtype=torch.float16, device=device)
      num_samples = num_samples or config.mc_samples
      temperature = temperature or config.mc_temperature
      
      model.train()  # 关键：强制激活dropout
      logits_list = []
      start_time = time.time()
      for _ in range(num_samples):
          with autocast(enabled=config.use_amp):
              outputs = model(images)
              logits = outputs['semantic'] / temperature
              logits_list.append(logits)
      total_time = time.time() - start_time
      
      logits_stack = torch.stack(logits_list, dim=0)  # [samples, B, C, X, Y, Z]
      probs = F.softmax(logits_stack, dim=2)
      mean_probs = probs.mean(0)
      pred = mean_probs.argmax(1)
      
      # 不确定性指标
      variance = probs.var(0).mean(1)      # 平均通道variance
      entropy = - (mean_probs * torch.log(mean_probs + 1e-6)).sum(1)
      
      print(f"MC Dropout: {num_samples} samples in {total_time:.2f}s (FPS equiv: {num_samples/total_time:.1f})")
      print(f"Mean Variance: {variance.mean().item():.6f} | Mean Entropy: {entropy.mean().item():.6f}")
      
      return {
          'pred': pred,
          'uncertainty_variance': variance,
          'uncertainty_entropy': entropy,
          'logits_mean': mean_probs
      }
  ```

  TORCHSPARSE_AVAILABLE = False
  try:
      import spconv.pytorch as spconv
      from spconv.pytorch import SparseConvTensor
      SPCONV_AVAILABLE = True
  except:
      pass

      import torchsparse
      from torchsparse import SparseTensor
      from torchsparse.nn import Conv3d, BatchNorm
      TORCHSPARSE_AVAILABLE = True
  except:
      pass

  class SparseConvBlock(nn.Module):
      def __init__(self, in_channels, out_channels, kernel_size=3):
          super().__init__()
          if SPCONV_AVAILABLE:
              self.conv = spconv.SubMConv3d(in_channels, out_channels, kernel_size, padding=kernel_size // 2, bias=False)
              self.bn = nn.BatchNorm1d(out_channels)
              self.backend = 'spconv'
          elif TORCHSPARSE_AVAILABLE:
              self.conv = Conv3d(in_channels, out_channels, kernel_size=3, stride=1)
              self.bn = BatchNorm(out_channels)
              self.backend = 'torchsparse'
          else:
              self.conv = PseudoSparseConv3d(in_channels, out_channels, kernel_size, padding=kernel_size // 2)
              self.bn = nn.BatchNorm3d(out_channels)
              self.backend = 'dense'
          self.act = nn.GELU()

      def forward(self, x):
          if self.backend == 'spconv':
              x = self.conv(x)
              x = x.replace_feature(self.act(self.bn(x.features)))
          elif self.backend == 'torchsparse':
              x = self.conv(x)
              x = self.bn(x)
              x = self.act(x)
          else:
              x = self.act(self.bn(self.conv(x)))
          return x
  ```
