```python
"""
全景语义分割传感器管理器
管理6个语义分割相机生成360°全景语义图
"""

import carla
import numpy as np
import weakref
from queue import Queue, Empty

from ..config.panorama_config import (
    PANO_WIDTH, PANO_HEIGHT, CUBE_SIZE, CUBE_FOV,
    CUBE_FACE_CONFIGS, PANO_LOCATION
)
from ..processing.panorama_tools import PanoramaTools


class PanoramaSemanticManager:
    """
    全景语义传感器管理器
    管理6个语义分割相机，生成360°全景语义图
    """

    def __init__(self, world, vehicle):
        """
        初始化全景语义管理器

        Args:
            world: carla.World对象
            vehicle: carla.Vehicle对象 (hero车辆)
        """
        self.world = world
        self.vehicle = vehicle
        self.bp_library = world.get_blueprint_library()

        # 传感器列表
        self.sensors = []
        self.queues = {}  # {'front': q, 'right': q, ...}

        # 工具
        self.pano_tools = PanoramaTools(PANO_WIDTH, PANO_HEIGHT, CUBE_SIZE)

        # 初始化
        self._setup_sensors()

    def _setup_sensors(self):
        """创建6个语义分割相机"""

        # 基础位置 (车顶中心)
        base_loc = carla.Location(
            x=PANO_LOCATION['x'],
            y=PANO_LOCATION['y'],
            z=PANO_LOCATION['z']
        )

        for face in CUBE_FACE_CONFIGS:
            name = face['name']
            rot = face['rot']

            transform = carla.Transform(
                base_loc,
                carla.Rotation(pitch=rot[0], yaw=rot[1], roll=rot[2])
            )

            # 语义分割相机
            sem_bp = self.bp_library.find('sensor.camera.semantic_segmentation')
            sem_bp.set_attribute('image_size_x', str(CUBE_SIZE))
            sem_bp.set_attribute('image_size_y', str(CUBE_SIZE))
            sem_bp.set_attribute('fov', str(CUBE_FOV))
            sem_bp.set_attribute('sensor_tick', '0.0')  # 同步模式

            sensor = self.world.spawn_actor(sem_bp, transform, attach_to=self.vehicle)
            self.sensors.append(sensor)

            # 队列
            queue = Queue()
            self.queues[name] = queue

            weak_queue = weakref.ref(queue)
            sensor.listen(lambda data, q=weak_queue: self._sensor_callback(data, q))

        print(f"✓ 全景语义相机已创建: 6个方向")

    def _sensor_callback(self, data, weak_queue):
        """传感器回调"""
        queue = weak_queue()
        if queue:
            queue.put(data)

    def get_semantic_panorama(self, timeout=2.0):
        """
        获取一帧全景语义图

        Args:
            timeout: 超时时间（秒）

        Returns:
            semantic_pano: (H, W) uint8 全景语义图
                          如果超时返回None
        """
        # 1. 获取所有面数据
        semantic_faces = []
        timestamp = 0

        try:
            for face in CUBE_FACE_CONFIGS:
                name = face['name']
                data = self.queues[name].get(timeout=timeout)

                # 解析语义图像
                array = np.frombuffer(data.raw_data, dtype=np.uint8)
                array = array.reshape((CUBE_SIZE, CUBE_SIZE, 4))
                semantic_face = array[:, :, 2]  # Red通道

                semantic_faces.append(semantic_face)
                timestamp = data.timestamp

        except Empty:
            print("⚠ 全景语义相机数据超时")
            return None

        # 2. 拼接为全景图
        semantic_pano = self.pano_tools.stitch(semantic_faces)

        return {
            'semantic_pano': semantic_pano,
            'timestamp': timestamp
        }

    def destroy(self):
        """销毁所有传感器"""
        for sensor in self.sensors:
            if sensor.is_alive:
                sensor.stop()
                sensor.destroy()
        self.sensors = []
        self.queues = {}
        print("✓ 全景语义相机已销毁")
```

### 步骤3: 移植全景拼接工具（精简版）

**新文件**: `carla_data_collection/processing/panorama_tools.py`

**修改内容**:
1. ✅ 保留 `__init__` 和 `_init_remap_tables` (预计算映射表)

**完整代码**:
```python
"""
全景图拼接工具
将6个CubeMap面拼接为Equirectangular全景图
"""

import numpy as np
import cv2


class PanoramaTools:
    """全景图拼接工具"""

    def __init__(self, pano_w, pano_h, cube_size):
        """
        初始化拼接工具

        Args:
            pano_w: 全景图宽度
            pano_h: 全景图高度
            cube_size: 立方体面尺寸
        """
        self.pano_w = pano_w
        self.pano_h = pano_h
        self.cube_size = cube_size
        self.map_x, self.map_y = self._init_remap_tables()

    def _init_remap_tables(self):
        """
        预计算从Equirectangular到CubeMap的映射表
        用于cv2.remap高效拼接
        """
        # 1. 生成全景图的像素坐标 (u, v)
        uv = np.indices((self.pano_h, self.pano_w), dtype=np.float32)
        u, v = uv[1], uv[0]

        # 2. 像素 → 球面角度
        theta = (u / self.pano_w) * 2 * np.pi    # [0, 2π]
        phi = (0.5 - v / self.pano_h) * np.pi    # [π/2, -π/2]

        # 3. 球面坐标 → 3D方向向量
        x = np.cos(phi) * np.cos(theta)
        y = np.cos(phi) * np.sin(theta)
        z = np.sin(phi)

        # 4. 3D向量 → CubeFace索引
        abs_x, abs_y, abs_z = np.abs(x), np.abs(y), np.abs(z)

        is_x_pos = (x > 0)
        is_y_pos = (y > 0)
        is_z_pos = (z > 0)

        # 判断主轴
        max_axis = np.maximum(np.maximum(abs_x, abs_y), abs_z)

        face_id = np.zeros_like(x, dtype=np.int8)
        u_cube = np.zeros_like(x)
        v_cube = np.zeros_like(x)

        # Front (X+)
        mask = (abs_x == max_axis) & is_x_pos
        face_id[mask] = 0
        u_cube[mask] = y[mask] / x[mask]
        v_cube[mask] = -z[mask] / x[mask]

        # Right (Y+)
        mask = (abs_y == max_axis) & is_y_pos
        face_id[mask] = 1
        u_cube[mask] = -x[mask] / y[mask]
        v_cube[mask] = -z[mask] / y[mask]

        # Back (X-)
        mask = (abs_x == max_axis) & ~is_x_pos
        face_id[mask] = 2
        u_cube[mask] = -y[mask] / -x[mask]
        v_cube[mask] = -z[mask] / -x[mask]

        # Left (Y-)
        mask = (abs_y == max_axis) & ~is_y_pos
        face_id[mask] = 3
        u_cube[mask] = x[mask] / -y[mask]
        v_cube[mask] = -z[mask] / -y[mask]

        # Up (Z+)
        mask = (abs_z == max_axis) & is_z_pos
        face_id[mask] = 4
        u_cube[mask] = y[mask] / z[mask]
        v_cube[mask] = x[mask] / z[mask]

        # Down (Z-)
        mask = (abs_z == max_axis) & ~is_z_pos
        face_id[mask] = 5
        u_cube[mask] = y[mask] / -z[mask]
        v_cube[mask] = -x[mask] / -z[mask]

        # 5. 归一化坐标 → 像素坐标
        map_x = (u_cube + 1.0) * 0.5 * (self.cube_size - 1)
        map_y = (v_cube + 1.0) * 0.5 * (self.cube_size - 1)

        # 6. 编码face ID
        map_x += face_id.astype(np.float32) * self.cube_size

        return map_x.astype(np.float32), map_y.astype(np.float32)

    def stitch(self, cube_faces):
        """
        将6个CubeMap面拼接为全景图

        Args:
            cube_faces: list of 6 images (Front, Right, Back, Left, Up, Down)
                        每个面形状 (H, W) 或 (H, W, C)

        Returns:
            panorama: (PANO_H, PANO_W) 或 (PANO_H, PANO_W, C)
        """
        # 水平拼接所有面
        if isinstance(cube_faces, list):
            atlas = np.concatenate(cube_faces, axis=1)
        else:
            atlas = cube_faces

        # 使用remap进行映射
        # 语义图用INTER_NEAREST防止插值产生不存在的类别
        if atlas.ndim == 2 or (atlas.ndim == 3 and atlas.shape[2] == 1):
            interp = cv2.INTER_NEAREST
        else:
            interp = cv2.INTER_LINEAR

        pano = cv2.remap(atlas, self.map_x, self.map_y,
                        interpolation=interp,
                        borderMode=cv2.BORDER_CONSTANT)
        return pano
```

#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
测试全景语义相机功能
"""

import sys
import os
from pathlib import Path

# 添加项目路径
project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))

import carla
import numpy as np
import cv2
import time


def test_panorama_semantic():
    """测试全景语义相机"""

    print("="*60)
    print("测试360°全景语义相机")
    print("="*60)

    # 连接CARLA
    client = carla.Client('localhost', 2000)
    client.set_timeout(10.0)
    world = client.get_world()
    print(f"✓ 已连接到CARLA: {world.get_map().name}")

    # 同步模式
    settings = world.get_settings()
    settings.synchronous_mode = True
    settings.fixed_delta_seconds = 0.05
    world.apply_settings(settings)

    vehicle = None
    pano_manager = None

    try:
        # 创建车辆
        bp_lib = world.get_blueprint_library()
        vehicle_bp = bp_lib.filter('vehicle.tesla.*')[0]
        spawn_points = world.get_map().get_spawn_points()
        vehicle = world.spawn_actor(vehicle_bp, spawn_points[0])
        print(f"✓ 车辆已生成: {vehicle.type_id}")

        # 创建全景语义管理器
        from sensors.panorama_semantic_manager import PanoramaSemanticManager
        pano_manager = PanoramaSemanticManager(world, vehicle)

        # 预热
        for _ in range(10):
            world.tick()

        # 采集一帧
        print("\n采集全景语义图...")
        world.tick()
        result = pano_manager.get_semantic_panorama(timeout=2.0)

        if result is None:
            print("❌ 采集失败")
            return

        semantic_pano = result['semantic_pano']
        timestamp = result['timestamp']

        print(f"✓ 采集成功")
        print(f"  全景图尺寸: {semantic_pano.shape}")
        print(f"  时间戳: {timestamp:.3f}")
        print(f"  语义类别: {np.unique(semantic_pano)}")

        # 保存可视化
        # CARLA语义颜色映射
        semantic_color = np.zeros((semantic_pano.shape[0], semantic_pano.shape[1], 3),
                                 dtype=np.uint8)
        # 简单可视化：每个类别一个颜色
        for label in np.unique(semantic_pano):
            mask = semantic_pano == label
            color = ((label * 37) % 255, (label * 73) % 255, (label * 127) % 255)
            semantic_color[mask] = color

        output_path = 'test_panorama_semantic.png'
        cv2.imwrite(output_path, semantic_color)
        print(f"✓ 已保存可视化: {output_path}")

    except Exception as e:
        print(f"❌ 错误: {e}")
        import traceback
        traceback.print_exc()

    finally:
        # 清理
        if pano_manager:
            pano_manager.destroy()
        if vehicle:
            vehicle.destroy()

        settings.synchronous_mode = False
        world.apply_settings(settings)

        print("\n测试完成")


if __name__ == '__main__':
    test_panorama_semantic()

## 五、常见问题

### Q1: 全景图有黑色条纹

**原因**: CubeMap面之间的接缝
**解决**: 正常现象，微小接缝（<1像素）可接受

### Q2: 某些面数据超时

**原因**: 同步模式下tick频率不够
**解决**: 增加预热帧数（从10帧到20帧）

### Q3: 语义颜色不一致

**原因**: CARLA语义标签ID直接用作颜色
**解决**: 使用正确的颜色映射表（见occupancy_config.py）
