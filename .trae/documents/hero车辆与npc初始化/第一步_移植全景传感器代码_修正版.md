## 二、完整的PanoramaManager

**文件**: `carla_data_collection/sensors/panorama_manager.py`

```python
"""
全景传感器管理器（完整版）
管理12个CubeMap相机：6个深度 + 6个语义
"""

import carla
import numpy as np
import weakref
from queue import Queue, Empty

from ..config.panorama_config import (
    PANO_WIDTH, PANO_HEIGHT, CUBE_SIZE, CUBE_FOV,
    CUBE_FACE_CONFIGS, PANO_LOCATION
)
from ..processing.panorama_tools import PanoramaTools


class PanoramaManager:
    """
    全景传感器管理器
    管理 6个方向 × 2种类型 = 12个传感器
    """

    def __init__(self, world, vehicle):
        self.world = world
        self.vehicle = vehicle
        self.bp_library = world.get_blueprint_library()

        # 传感器列表
        self.sensors = []
        self.queues = {}  # {'front_depth': q, 'front_semantic': q, ...}

        # 工具
        self.pano_tools = PanoramaTools(PANO_WIDTH, PANO_HEIGHT, CUBE_SIZE)

        # 初始化
        self._setup_sensors()

    def _setup_sensors(self):
        """创建12个传感器"""

        # 基础位置（车顶中心）
        base_loc = carla.Location(
            x=PANO_LOCATION['x'],
            y=PANO_LOCATION['y'],
            z=PANO_LOCATION['z']
        )

        for face in CUBE_FACE_CONFIGS:
            name = face['name']
            rot = face['rot']

            transform = carla.Transform(
                base_loc,
                carla.Rotation(pitch=rot[0], yaw=rot[1], roll=rot[2])
            )

            # 1. 深度相机 ✅
            self._spawn_depth_camera(name, transform)

            # 2. 语义相机 ✅
            self._spawn_semantic_camera(name, transform)

        print(f"✓ 全景传感器已创建: 6深度 + 6语义 = 12个")

    def _spawn_depth_camera(self, face_name, transform):
        """创建深度相机"""
        depth_bp = self.bp_library.find('sensor.camera.depth')
        depth_bp.set_attribute('image_size_x', str(CUBE_SIZE))
        depth_bp.set_attribute('image_size_y', str(CUBE_SIZE))
        depth_bp.set_attribute('fov', str(CUBE_FOV))
        depth_bp.set_attribute('sensor_tick', '0.0')

        sensor = self.world.spawn_actor(depth_bp, transform, attach_to=self.vehicle)
        self.sensors.append(sensor)

        queue = Queue()
        self.queues[f"{face_name}_depth"] = queue

        weak_queue = weakref.ref(queue)
        sensor.listen(lambda data, q=weak_queue: self._sensor_callback(data, q))

    def _spawn_semantic_camera(self, face_name, transform):
        """创建语义相机"""
        sem_bp = self.bp_library.find('sensor.camera.semantic_segmentation')
        sem_bp.set_attribute('image_size_x', str(CUBE_SIZE))
        sem_bp.set_attribute('image_size_y', str(CUBE_SIZE))
        sem_bp.set_attribute('fov', str(CUBE_FOV))
        sem_bp.set_attribute('sensor_tick', '0.0')

        sensor = self.world.spawn_actor(sem_bp, transform, attach_to=self.vehicle)
        self.sensors.append(sensor)

        queue = Queue()
        self.queues[f"{face_name}_semantic"] = queue

        weak_queue = weakref.ref(queue)
        sensor.listen(lambda data, q=weak_queue: self._sensor_callback(data, q))

    def _sensor_callback(self, data, weak_queue):
        """传感器回调"""
        queue = weak_queue()
        if queue:
            queue.put(data)

    def get_panorama_data(self, timeout=2.0):
        """
        获取一帧全景数据

        Returns:
            dict: {
                'depth_pano': (H, W) float32,
                'semantic_pano': (H, W) uint8,
                'timestamp': float
            }
        """
        # 获取所有面数据
        depth_faces = []
        semantic_faces = []
        timestamp = 0

        try:
            for face in CUBE_FACE_CONFIGS:
                name = face['name']

                # 深度
                d_data = self.queues[f"{name}_depth"].get(timeout=timeout)
                d_array = np.frombuffer(d_data.raw_data, dtype=np.uint8)
                d_array = d_array.reshape((CUBE_SIZE, CUBE_SIZE, 4))

                # 解码深度
                R = d_array[:, :, 2].astype(np.float32)
                G = d_array[:, :, 1].astype(np.float32)
                B = d_array[:, :, 0].astype(np.float32)
                normalized = (R + G * 256 + B * 65536) / (256**3 - 1)
                depth_meters = normalized * 1000.0  # 单位：米
                depth_faces.append(depth_meters)

                # 语义
                s_data = self.queues[f"{name}_semantic"].get(timeout=timeout)
                s_array = np.frombuffer(s_data.raw_data, dtype=np.uint8)
                s_array = s_array.reshape((CUBE_SIZE, CUBE_SIZE, 4))
                semantic_faces.append(s_array[:, :, 2])  # Red通道

                timestamp = d_data.timestamp

        except Empty:
            print("⚠ 全景传感器数据超时")
            return None

        # 拼接
        depth_pano = self.pano_tools.stitch(depth_faces)
        semantic_pano = self.pano_tools.stitch(semantic_faces)

        return {
            'depth_pano': depth_pano,
            'semantic_pano': semantic_pano,
            'timestamp': timestamp
        }

    def destroy(self):
        """销毁所有传感器"""
        for sensor in self.sensors:
            if sensor.is_alive:
                sensor.stop()
                sensor.destroy()
        self.sensors = []
        self.queues = {}
        print("✓ 全景传感器已销毁")
```

## 三、测试脚本

**文件**: `carla_data_collection/scripts/test_panorama.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""测试全景传感器（深度+语义）"""

import sys
from pathlib import Path
project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))

import carla
import numpy as np
import cv2

def test_panorama():
    client = carla.Client('localhost', 2000)
    client.set_timeout(10.0)
    world = client.get_world()

    # 同步模式
    settings = world.get_settings()
    settings.synchronous_mode = True
    settings.fixed_delta_seconds = 0.05
    world.apply_settings(settings)

    vehicle = None
    pano_manager = None

    try:
        # 创建车辆
        bp_lib = world.get_blueprint_library()
        vehicle_bp = bp_lib.filter('vehicle.tesla.*')[0]
        spawn_points = world.get_map().get_spawn_points()
        vehicle = world.spawn_actor(vehicle_bp, spawn_points[0])

        # 创建全景管理器
        from sensors.panorama_manager import PanoramaManager
        pano_manager = PanoramaManager(world, vehicle)

        # 预热
        for _ in range(10):
            world.tick()

        # 采集
        print("\n采集全景数据...")
        world.tick()
        result = pano_manager.get_panorama_data(timeout=2.0)

        if result is None:
            print("❌ 采集失败")
            return

        depth_pano = result['depth_pano']
        semantic_pano = result['semantic_pano']

        print(f"✓ 采集成功")
        print(f"  深度全景: {depth_pano.shape}, 范围: [{depth_pano.min():.1f}, {depth_pano.max():.1f}]m")
        print(f"  语义全景: {semantic_pano.shape}, 类别: {np.unique(semantic_pano)}")

        # 保存可视化
        depth_vis = (depth_pano / 100 * 255).astype(np.uint8)
        cv2.imwrite('test_pano_depth.png', depth_vis)

        semantic_vis = (semantic_pano * 10 % 256).astype(np.uint8)
        cv2.imwrite('test_pano_semantic.png', semantic_vis)

        print("✓ 已保存: test_pano_depth.png, test_pano_semantic.png")

    finally:
        if pano_manager:
            pano_manager.destroy()
        if vehicle:
            vehicle.destroy()

        settings.synchronous_mode = False
        world.apply_settings(settings)


if __name__ == '__main__':
    test_panorama()
```
