### 1.1 主训练脚本

```python
# training/trainer.py

import torch
import torch.nn as nn
import torch.distributed as dist
from torch.nn.parallel import DistributedDataParallel as DDP
from torch.utils.data import DataLoader, DistributedSampler
from torch.cuda.amp import autocast, GradScaler
import wandb
import yaml
import os
from pathlib import Path
from tqdm import tqdm
import numpy as np

from models.hydranet import TeslaHydraNet
from dataset.builder import CARLAHydraDataset
from losses.multi_task_loss import HydraMultiTaskLoss


class HydraTrainer:
    """
    九头蛇网络训练器

    支持功能:
    1. 分布式训练 (DDP)
    2. 混合精度训练 (FP16)
    3. 梯度累积
    4. WandB 实验追踪
    5. 早停和检查点保存
    """

    def __init__(self, config_path: str):
        # 加载配置
        with open(config_path, 'r', encoding='utf-8') as f:
            self.config = yaml.safe_load(f)

        # 设置分布式训练
        self.local_rank = int(os.environ.get('LOCAL_RANK', 0))
        self.world_size = int(os.environ.get('WORLD_SIZE', 1))
        self.is_distributed = self.world_size > 1

        if self.is_distributed:
            dist.init_process_group(backend='nccl')
            torch.cuda.set_device(self.local_rank)

        self.device = torch.device(f'cuda:{self.local_rank}')
        self.is_main_process = self.local_rank == 0

        # 初始化 WandB (仅主进程)
        if self.is_main_process and self.config['logging']['use_wandb']:
            wandb.init(
                project=self.config['logging']['wandb_project'],
                config=self.config,
                name=f"hydra_{self.config['model']['backbone']}"
            )

        # 构建模型
        self.model = self._build_model()

        # 构建数据加载器
        self.train_loader, self.val_loader = self._build_dataloaders()

        # 构建优化器和调度器
        self.optimizer = self._build_optimizer()
        self.scheduler = self._build_scheduler()

        # 构建损失函数
        self.criterion = HydraMultiTaskLoss(device=self.device)

        # 混合精度训练
        self.scaler = GradScaler() if self.config['training']['mixed_precision'] else None

        # 训练状态
        self.current_epoch = 0
        self.global_step = 0
        self.best_val_loss = float('inf')
        self.patience_counter = 0

        # 检查点目录
        self.checkpoint_dir = Path(self.config['logging']['save_dir'])
        self.checkpoint_dir.mkdir(parents=True, exist_ok=True)

    def _build_model(self):
        """构建模型"""
        model = TeslaHydraNet(
            backbone=self.config['model']['backbone'],
            pretrained=self.config['model']['pretrained'],
            bev_size=self.config['model']['bev_size'],
            feature_dim=self.config['model']['feature_dim'],
            num_cameras=self.config['model']['num_cameras']
        ).to(self.device)

        # 分布式训练包装
        if self.is_distributed:
            model = DDP(
                model,
                device_ids=[self.local_rank],
                output_device=self.local_rank,
                find_unused_parameters=True
            )

        if self.is_main_process:
            num_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
            print(f"✓ 模型已构建: {num_params / 1e6:.2f}M 参数")

        return model

    def _build_dataloaders(self):
        """构建数据加载器"""
        # 训练集
        train_dataset = CARLAHydraDataset(
            data_root=self.config['data']['root_dir'],
            split='train',
            augment=True
        )

        # 验证集
        val_dataset = CARLAHydraDataset(
            data_root=self.config['data']['root_dir'],
            split='val',
            augment=False
        )

        # 分布式采样器
        train_sampler = DistributedSampler(
            train_dataset,
            num_replicas=self.world_size,
            rank=self.local_rank,
            shuffle=True
        ) if self.is_distributed else None

        val_sampler = DistributedSampler(
            val_dataset,
            num_replicas=self.world_size,
            rank=self.local_rank,
            shuffle=False
        ) if self.is_distributed else None

        # 数据加载器
        train_loader = DataLoader(
            train_dataset,
            batch_size=self.config['data']['batch_size'],
            sampler=train_sampler,
            shuffle=(train_sampler is None),
            num_workers=self.config['data']['num_workers'],
            pin_memory=self.config['data']['pin_memory'],
            prefetch_factor=self.config['data']['prefetch_factor'],
            persistent_workers=True
        )

        val_loader = DataLoader(
            val_dataset,
            batch_size=self.config['data']['batch_size'],
            sampler=val_sampler,
            shuffle=False,
            num_workers=self.config['data']['num_workers'],
            pin_memory=self.config['data']['pin_memory']
        )

        if self.is_main_process:
            print(f"✓ 数据集已加载: 训练 {len(train_dataset)} | 验证 {len(val_dataset)}")

        return train_loader, val_loader

    def _build_optimizer(self):
        """构建优化器"""
        opt_config = self.config['training']['optimizer']

        if opt_config['type'] == 'AdamW':
            optimizer = torch.optim.AdamW(
                self.model.parameters(),
                lr=opt_config['lr'],
                weight_decay=opt_config['weight_decay'],
                betas=opt_config['betas']
            )
        else:
            raise ValueError(f"不支持的优化器: {opt_config['type']}")

        return optimizer

    def _build_scheduler(self):
        """构建学习率调度器"""
        sched_config = self.config['training']['lr_scheduler']

        if sched_config['type'] == 'CosineAnnealingWarmRestarts':
            scheduler = torch.optim.lr_scheduler.CosineAnnealingWarmRestarts(
                self.optimizer,
                T_0=sched_config['T_0'],
                T_mult=sched_config['T_mult'],
                eta_min=sched_config['eta_min']
            )
        else:
            raise ValueError(f"不支持的调度器: {sched_config['type']}")

        return scheduler

    def train_epoch(self):
        """训练一个 epoch"""
        self.model.train()

        if self.is_distributed:
            self.train_loader.sampler.set_epoch(self.current_epoch)

        epoch_losses = []
        grad_accum_steps = self.config['training']['gradient_accumulation_steps']

        pbar = tqdm(
            self.train_loader,
            desc=f"Epoch {self.current_epoch}",
            disable=not self.is_main_process
        )

        for batch_idx, batch in enumerate(pbar):
            # 移动到 GPU
            cameras = batch['cameras'].to(self.device)
            gps = batch['gps'].to(self.device)
            imu = batch['imu'].to(self.device)
            speed = batch['speed'].to(self.device)
            labels = {k: v.to(self.device) for k, v in batch['labels'].items()}

            # 前向传播 (混合精度)
            with autocast(enabled=self.scaler is not None):
                outputs = self.model(cameras, gps, imu, speed)
                loss_dict = self.criterion(outputs, labels)
                loss = loss_dict['total_loss']

                # 梯度累积
                loss = loss / grad_accum_steps

            # 反向传播
            if self.scaler is not None:
                self.scaler.scale(loss).backward()
            else:
                loss.backward()

            # 梯度更新
            if (batch_idx + 1) % grad_accum_steps == 0:
                # 梯度裁剪
                if self.scaler is not None:
                    self.scaler.unscale_(self.optimizer)

                torch.nn.utils.clip_grad_norm_(
                    self.model.parameters(),
                    self.config['training']['gradient_clip_norm']
                )

                # 优化器步进
                if self.scaler is not None:
                    self.scaler.step(self.optimizer)
                    self.scaler.update()
                else:
                    self.optimizer.step()

                self.optimizer.zero_grad()
                self.global_step += 1

            # 记录损失
            epoch_losses.append(loss.item() * grad_accum_steps)

            # 更新进度条
            pbar.set_postfix({
                'loss': f"{loss.item() * grad_accum_steps:.4f}",
                'lr': f"{self.optimizer.param_groups[0]['lr']:.6f}"
            })

            # 日志记录
            if self.global_step % self.config['logging']['log_interval'] == 0 and self.is_main_process:
                log_dict = {
                    'train/total_loss': loss.item() * grad_accum_steps,
                    'train/learning_rate': self.optimizer.param_groups[0]['lr'],
                    'train/epoch': self.current_epoch,
                    'train/step': self.global_step
                }

                # 记录各任务损失
                for task_name, task_loss in loss_dict['losses'].items():
                    log_dict[f'train/loss_{task_name}'] = task_loss.item()

                # 记录任务权重
                for task_name, weight in loss_dict['weights'].items():
                    log_dict[f'train/weight_{task_name}'] = weight

                if wandb.run is not None:
                    wandb.log(log_dict, step=self.global_step)

        # 学习率调度
        self.scheduler.step()

        avg_loss = np.mean(epoch_losses)
        return avg_loss

    @torch.no_grad()
    def validate(self):
        """验证"""
        self.model.eval()

        val_losses = []

        pbar = tqdm(
            self.val_loader,
            desc="Validation",
            disable=not self.is_main_process
        )

        for batch in pbar:
            # 移动到 GPU
            cameras = batch['cameras'].to(self.device)
            gps = batch['gps'].to(self.device)
            imu = batch['imu'].to(self.device)
            speed = batch['speed'].to(self.device)
            labels = {k: v.to(self.device) for k, v in batch['labels'].items()}

            # 前向传播
            outputs = self.model(cameras, gps, imu, speed)
            loss_dict = self.criterion(outputs, labels)
            loss = loss_dict['total_loss']

            val_losses.append(loss.item())

            pbar.set_postfix({'loss': f"{loss.item():.4f}"})

        avg_val_loss = np.mean(val_losses)

        if self.is_main_process and wandb.run is not None:
            wandb.log({
                'val/loss': avg_val_loss,
                'val/epoch': self.current_epoch
            }, step=self.global_step)

        return avg_val_loss

    def save_checkpoint(self, filename='checkpoint.pth', is_best=False):
        """保存检查点"""
        if not self.is_main_process:
            return

        checkpoint = {
            'epoch': self.current_epoch,
            'global_step': self.global_step,
            'model_state_dict': self.model.module.state_dict() if self.is_distributed else self.model.state_dict(),
            'optimizer_state_dict': self.optimizer.state_dict(),
            'scheduler_state_dict': self.scheduler.state_dict(),
            'best_val_loss': self.best_val_loss,
            'config': self.config
        }

        checkpoint_path = self.checkpoint_dir / filename
        torch.save(checkpoint, checkpoint_path)
        print(f"✓ 检查点已保存: {checkpoint_path}")

        if is_best:
            best_path = self.checkpoint_dir / 'best_model.pth'
            torch.save(checkpoint, best_path)
            print(f"✓ 最佳模型已保存: {best_path}")

    def train(self):
        """完整训练流程"""
        num_epochs = self.config['training']['num_epochs']
        patience = self.config['training']['early_stopping']['patience']
        min_delta = self.config['training']['early_stopping']['min_delta']

        if self.is_main_process:
            print("\n" + "="*80)
            print("开始训练九头蛇网络")
            print("="*80)

        for epoch in range(num_epochs):
            self.current_epoch = epoch

            # 训练
            train_loss = self.train_epoch()

            # 验证
            val_loss = self.validate()

            if self.is_main_process:
                print(f"\nEpoch {epoch}: Train Loss = {train_loss:.4f}, Val Loss = {val_loss:.4f}")

            # 保存检查点
            if (epoch + 1) % self.config['logging']['save_interval'] == 0:
                self.save_checkpoint(f'checkpoint_epoch_{epoch}.pth')

            # 早停检查
            if val_loss < self.best_val_loss - min_delta:
                self.best_val_loss = val_loss
                self.patience_counter = 0
                self.save_checkpoint(is_best=True)
            else:
                self.patience_counter += 1

            if self.patience_counter >= patience:
                if self.is_main_process:
                    print(f"\n早停触发: {patience} 个 epoch 无改善")
                break

        if self.is_main_process:
            print("\n" + "="*80)
            print("训练完成!")
            print(f"最佳验证损失: {self.best_val_loss:.4f}")
            print("="*80)

        # 清理分布式训练
        if self.is_distributed:
            dist.destroy_process_group()


# ===== 使用示例 =====
if __name__ == '__main__':
    import argparse

    parser = argparse.ArgumentParser()
    parser.add_argument('--config', type=str, default='training/config.yaml')
    args = parser.parse_args()

    trainer = HydraTrainer(args.config)
    trainer.train()
```

### 2.1 CARLA 自动驾驶 Agent

```python
# deployment/carla_agent.py

import carla
import torch
import numpy as np
import time
from pathlib import Path
from collections import deque

from models.hydranet import TeslaHydraNet
from carla_interface.sensors.camera_array import CameraArray
from carla_interface.sensors.sensor_config import SensorSuite


class HydraAutonomousAgent:
    """
    九头蛇自动驾驶 Agent

    功能:
    1. 实时推理 (<100ms)
    2. 传感器数据缓冲
    3. 控制指令平滑
    4. 安全检查
    """

    def __init__(
        self,
        model_path: str,
        host='localhost',
        port=2000,
        device='cuda'
    ):
        # CARLA 连接
        self.client = carla.Client(host, port)
        self.client.set_timeout(10.0)
        self.world = self.client.get_world()

        # 加载模型
        self.device = torch.device(device)
        checkpoint = torch.load(model_path, map_location=self.device)

        self.model = TeslaHydraNet(
            backbone=checkpoint['config']['model']['backbone'],
            pretrained=False,
            bev_size=checkpoint['config']['model']['bev_size'],
            feature_dim=checkpoint['config']['model']['feature_dim'],
            num_cameras=checkpoint['config']['model']['num_cameras']
        ).to(self.device)

        self.model.load_state_dict(checkpoint['model_state_dict'])
        self.model.eval()

        print(f"✓ 模型已加载: {model_path}")

        # 车辆和传感器
        self.vehicle = None
        self.camera_array = None

        # 数据缓冲
        self.sensor_buffer = {
            'cameras': None,
            'gps': None,
            'imu': None,
            'speed': 0.0
        }

        # 控制平滑
        self.steering_history = deque(maxlen=5)
        self.throttle_history = deque(maxlen=3)

        # 统计信息
        self.frame_count = 0
        self.total_inference_time = 0.0

    def setup_vehicle(self):
        """生成车辆"""
        bp_library = self.world.get_blueprint_library()
        vehicle_bp = bp_library.find('vehicle.tesla.model3')

        spawn_points = self.world.get_map().get_spawn_points()
        spawn_point = np.random.choice(spawn_points)

        self.vehicle = self.world.spawn_actor(vehicle_bp, spawn_point)
        print(f"✓ 车辆已生成")

    def setup_sensors(self):
        """设置传感器"""
        sensor_suite = SensorSuite()

        # 相机阵列
        self.camera_array = CameraArray(
            self.world,
            self.vehicle,
            sensor_suite.cameras
        )

        # GPS
        gnss_bp = self.world.get_blueprint_library().find('sensor.other.gnss')
        self.gnss_sensor = self.world.spawn_actor(
            gnss_bp,
            carla.Transform(),
            attach_to=self.vehicle
        )
        self.gnss_sensor.listen(lambda data: self.sensor_buffer.update({
            'gps': torch.tensor([data.latitude, data.longitude, data.altitude], dtype=torch.float32)
        }))

        # IMU
        imu_bp = self.world.get_blueprint_library().find('sensor.other.imu')
        self.imu_sensor = self.world.spawn_actor(
            imu_bp,
            carla.Transform(),
            attach_to=self.vehicle
        )
        self.imu_sensor.listen(lambda data: self.sensor_buffer.update({
            'imu': torch.tensor([
                data.accelerometer.x, data.accelerometer.y, data.accelerometer.z,
                data.gyroscope.x, data.gyroscope.y, data.gyroscope.z
            ], dtype=torch.float32)
        }))

        print(f"✓ 传感器已启用")

    @torch.no_grad()
    def run_inference(self):
        """运行推理"""
        # 获取相机数据
        camera_frames = self.camera_array.get_latest_frame(timeout=0.5)
        if camera_frames is None:
            return None

        # 预处理
        cameras_list = []
        for cam_name in ['front_narrow', 'front_main', 'front_wide',
                         'left_front', 'left_rear', 'right_front', 'right_rear', 'rear']:
            img = camera_frames[cam_name]
            img_tensor = torch.from_numpy(img).permute(2, 0, 1).float() / 255.0

            # ImageNet 归一化
            mean = torch.tensor([0.485, 0.456, 0.406]).view(3, 1, 1)
            std = torch.tensor([0.229, 0.224, 0.225]).view(3, 1, 1)
            img_tensor = (img_tensor - mean) / std

            cameras_list.append(img_tensor)

        cameras = torch.stack(cameras_list, dim=0).unsqueeze(0).to(self.device)  # (1, 8, 3, H, W)

        # 获取其他传感器数据
        gps = self.sensor_buffer['gps'].unsqueeze(0).to(self.device) if self.sensor_buffer['gps'] is not None else torch.zeros(1, 3).to(self.device)
        imu = self.sensor_buffer['imu'].unsqueeze(0).to(self.device) if self.sensor_buffer['imu'] is not None else torch.zeros(1, 6).to(self.device)

        # 获取速度
        velocity = self.vehicle.get_velocity()
        speed = np.linalg.norm([velocity.x, velocity.y, velocity.z]) * 3.6  # km/h
        speed_tensor = torch.tensor([[speed]], dtype=torch.float32).to(self.device)

        # 推理
        start_time = time.time()
        outputs = self.model(cameras, gps, imu, speed_tensor)
        inference_time = (time.time() - start_time) * 1000  # ms

        self.total_inference_time += inference_time
        self.frame_count += 1

        return outputs, inference_time

    def postprocess_control(self, outputs):
        """后处理输出为控制指令"""
        # 提取控制输出
        steering = outputs['steering'][0].item()
        speed_target = outputs['speed'][0].item()
        brake_prob = outputs['brake'][0].item()

        # 平滑转向角
        self.steering_history.append(steering)
        steering_smooth = np.mean(self.steering_history)

        # 速度控制 (简单 PID)
        velocity = self.vehicle.get_velocity()
        current_speed = np.linalg.norm([velocity.x, velocity.y, velocity.z])  # m/s

        speed_error = speed_target - current_speed

        if speed_error > 0:
            throttle = np.clip(speed_error / 5.0, 0, 1)
            brake = 0.0
        else:
            throttle = 0.0
            brake = np.clip(-speed_error / 5.0, 0, 1)

        # 紧急刹车
        if brake_prob > 0.8:
            throttle = 0.0
            brake = 1.0

        # 平滑油门
        self.throttle_history.append(throttle)
        throttle_smooth = np.mean(self.throttle_history)

        return carla.VehicleControl(
            steer=float(np.clip(steering_smooth, -1, 1)),
            throttle=float(np.clip(throttle_smooth, 0, 1)),
            brake=float(np.clip(brake, 0, 1)),
            hand_brake=False,
            reverse=False
        )

    def run(self, duration=300):
        """运行自动驾驶"""
        print("\n" + "="*80)
        print("开始自动驾驶测试")
        print("="*80)

        start_time = time.time()

        try:
            while time.time() - start_time < duration:
                # 推理
                result = self.run_inference()
                if result is None:
                    time.sleep(0.01)
                    continue

                outputs, inference_time = result

                # 控制
                control = self.postprocess_control(outputs)
                self.vehicle.apply_control(control)

                # 打印统计
                if self.frame_count % 30 == 0:
                    avg_inference_time = self.total_inference_time / self.frame_count
                    velocity = self.vehicle.get_velocity()
                    speed = np.linalg.norm([velocity.x, velocity.y, velocity.z]) * 3.6

                    print(f"帧 {self.frame_count:5d} | "
                          f"推理: {inference_time:5.1f}ms (平均: {avg_inference_time:5.1f}ms) | "
                          f"速度: {speed:5.1f} km/h | "
                          f"转向: {control.steer:+.3f} | "
                          f"油门: {control.throttle:.3f} | "
                          f"刹车: {control.brake:.3f}")

                # 避免 CPU 100%
                time.sleep(0.01)

        except KeyboardInterrupt:
            print("\n用户中断")

        finally:
            self.cleanup()

        print("\n" + "="*80)
        print(f"测试完成! 总帧数: {self.frame_count} | 平均推理时间: {self.total_inference_time / self.frame_count:.1f}ms")
        print("="*80)

    def cleanup(self):
        """清理资源"""
        if self.camera_array:
            self.camera_array.destroy()
        if hasattr(self, 'gnss_sensor') and self.gnss_sensor:
            self.gnss_sensor.destroy()
        if hasattr(self, 'imu_sensor') and self.imu_sensor:
            self.imu_sensor.destroy()
        if self.vehicle:
            self.vehicle.destroy()

        print("✓ 资源已清理")
```

## 3. 持续学习迭代流程 {#持续学习}

### 3.1 从训练模型收集新数据

```python
# scripts/iterative_training.py

"""
迭代训练流程

循环:
  1. 使用当前模型在 CARLA 中自动驾驶
  2. 收集新数据 (特别是边缘案例)
  3. 合并到训练集
  4. 重新训练模型
  5. 评估性能
"""

import subprocess
import time
from pathlib import Path
import shutil


class IterativeTrainingPipeline:
    """迭代训练管道"""

    def __init__(
        self,
        data_root='./data',
        checkpoint_dir='./checkpoints',
        max_iterations=10,
        episodes_per_iteration=10
    ):
        self.data_root = Path(data_root)
        self.checkpoint_dir = Path(checkpoint_dir)
        self.max_iterations = max_iterations
        self.episodes_per_iteration = episodes_per_iteration

        # 当前最佳模型
        self.current_best_model = None

    def collect_data(self, iteration: int):
        """收集数据"""
        print(f"\n{'='*80}")
        print(f"迭代 {iteration}: 收集数据")
        print(f"{'='*80}")

        for episode in range(self.episodes_per_iteration):
            print(f"\n收集 Episode {episode+1}/{self.episodes_per_iteration}")

            # 启动数据采集
            cmd = [
                'python', 'carla_interface/data_collector.py',
                '--duration', '300',  # 5分钟
                '--output_dir', str(self.data_root / 'raw'),
                '--filename', f'iter{iteration}_ep{episode}.h5'
            ]

            # 如果有训练好的模型,使用模型驾驶
            if self.current_best_model:
                cmd.extend(['--use_model', str(self.current_best_model)])
            else:
                # 第一次迭代使用自动驾驶
                cmd.extend(['--use_autopilot'])

            subprocess.run(cmd, check=True)

        print(f"\n✓ 迭代 {iteration} 数据采集完成")

    def update_dataset_splits(self):
        """更新数据集划分"""
        print("\n更新数据集划分...")
        subprocess.run([
            'python', 'dataset/split_dataset.py',
            '--data_root', str(self.data_root)
        ], check=True)
        print("✓ 数据集划分完成")

    def train_model(self, iteration: int):
        """训练模型"""
        print(f"\n{'='*80}")
        print(f"迭代 {iteration}: 训练模型")
        print(f"{'='*80}")

        # 启动分布式训练
        subprocess.run([
            'bash', 'scripts/train_distributed.sh'
        ], check=True)

        # 更新当前最佳模型
        best_model = self.checkpoint_dir / 'best_model.pth'
        if best_model.exists():
            new_best = self.checkpoint_dir / f'best_model_iter{iteration}.pth'
            shutil.copy(best_model, new_best)
            self.current_best_model = new_best
            print(f"✓ 最佳模型已保存: {new_best}")

    def evaluate_model(self, iteration: int):
        """评估模型"""
        print(f"\n{'='*80}")
        print(f"迭代 {iteration}: 评估模型")
        print(f"{'='*80}")

        # 启动闭环测试
        subprocess.run([
            'python', 'deployment/carla_agent.py',
            '--model', str(self.current_best_model),
            '--duration', '600'  # 10分钟测试
        ], check=True)

        print(f"\n✓ 迭代 {iteration} 评估完成")

    def run(self):
        """运行完整迭代流程"""
        print("\n" + "="*80)
        print("开始迭代训练管道")
        print("="*80)

        for iteration in range(self.max_iterations):
            print(f"\n\n{'#'*80}")
            print(f"# 迭代 {iteration + 1} / {self.max_iterations}")
            print(f"{'#'*80}")

            # 1. 收集数据
            self.collect_data(iteration)

            # 2. 更新数据集
            self.update_dataset_splits()

            # 3. 训练模型
            self.train_model(iteration)

            # 4. 评估模型
            self.evaluate_model(iteration)

            # 等待用户确认
            input("\n按 Enter 继续下一次迭代...")

        print("\n" + "="*80)
        print("迭代训练完成!")
        print("="*80)


if __name__ == '__main__':
    pipeline = IterativeTrainingPipeline(
        data_root='./data',
        checkpoint_dir='./checkpoints',
        max_iterations=10,
        episodes_per_iteration=10
    )

    pipeline.run()
```
